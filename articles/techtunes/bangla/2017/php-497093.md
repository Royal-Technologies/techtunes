আসসালামু আলাইকুম, আশা করি সবাই ভাল আছেন। আমিও ভাল আছি। আজকে আমি পিএইচপি এর OOP টপিক নিয়ে আলোচনা করব। আজকে আমরা OOP এ Introduction, Classes এবং Constructor সম্পর্কে জানব। আমি আগে কয়েকটি পর্বে প্রোগ্রামগুলোর কোড পর্বগুলোর সাথে দিয়ে দিয়েছি। পাশাপাশি বুঝার জন্য আমার তৈরি করা ভিডিওর লিঙ্ক দিয়ে দিয়েছি। ত আজকেও আমি বিষয় সম্পর্কিত কোডগুলো আপনাদেরকে দিয়ে দিব। আর আপনারা বুজতে চাইলে আমার তৈরিকৃত ভিডিওগুলো দেখতে পারেন :)।

Classes পর্বের কোডঃ

```
<?php

class ourclass {

public $ob1 = "This is properties of ob1";

public $ob2 = "The value of object is 15";

function newfunction ()

{

echo $this -> ob1 .'</br>'. $this -> ob2;

}

} 

$object = new ourclass;

echo $object -> ob1.'</br>';

echo $object -> newfunction();

?>
```

Constructor পর্বের কোডঃ

```
<?php

class ourclass {

public $name, $value;

public function __construct($name, $value) 

{
    $this -> name =  $name;

    $this -> value = $value;

}
}

$object = new ourclass('yourname', '15');

echo $object -> name .' '. $object -> value;
```
