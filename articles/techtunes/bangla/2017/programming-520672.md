আজ আমরা নতুন একটি অধ্যায় নিয়ে আলচনা করব এবং এরিথমেটিক অপারেটর সিখব।

## অপারেটর কি

যে সেম্বল এর মাধ্যমে আমরা কন অপারেশন সম্পাদন করি সেটিয় হল অপারেটর।

উদাহরণ +,- ইত্যাদি।

## অপারেন্ড কি

জার উপর অপারেশন করা হয় সেটিকেই বলা হয় অপারেন্ড।

উদাহরণ ১০+২০ বা a+b  এখানে + হল অপারেটর এবং ১০ এবং ২০ অথবা  a এবং  b হল অপারেন্ড।

## এরিথমেটিক অপারাটর

+,-,\*,/,% এ ৫ টি হল এরিথমেটিক অপারেটর।

 public class Arithmatic {  
 public static void main(String[] args) {

//Declare variable

int a,b,sum,sub,mul,div,rem;

//initialize variable

a=21;  
 b=10;

//operation on variable

sum=a+b;  
 sub=a-b;  
 mul=a\*b;  
 div=a/b;  
 rem=a%b;

//show variable

System.out.println(sum);  
 System.out.println(sub);  
 System.out.println(mul);  
 System.out.println(div);  
 System.out.println(rem);

//show variable another way

System.out.println(a+b);  
 System.out.println(a-b);  
 System.out.println(a\*b);  
 System.out.println(a/b);  
 System.out.println(a%b);

}  
 }
